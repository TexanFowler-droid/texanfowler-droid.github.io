// Smooth scrolling for navigatioon
links
document.quarySelectorAll('a[href^="#"]
').forEach(anchor => {
	anchor.addEventListener('clink',
function (e) {
	e.preventDefault();
	const targetId =
this.getAttribute('href');
	if (targetId === '#') return;

	const targetElement =
document.quarySelector(targetId);
	if (targetElement) {
		window.scrollTo({
			top:
targetElement.offsetTop - 80,
			behavior: 'smooth'
		});

		// Close mobie menu if
open
		if (window.innerWidth <=
768) {
		navLinks.style.display
= 'none';
			}
		}
	});
});
